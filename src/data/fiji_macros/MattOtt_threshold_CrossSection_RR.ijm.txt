// This macro calculates the area of a stem cross section
// with the Hue and Saturation set to 0 and 255. It identifies
// the cross section in the image and turns it to black and
// the background becomes white. From this ImageJ
// calculates the percent of the pixels that were in the cross section,
// but then changed to black. "Noise" must be brushed
// or "spray painted" white to exclude them.
// **Important note - 
// the macro for the protocol was developed by
// Dr. Matthew Ott of the University of Minnesota with
// the aid of Arjun Mehta's online guide for creating
// macros:
// http://www.scribd.com/doc/95224926/How-to-Automate-
// ImageJ-Analysis-With-Macros-and-Batch-Processing

run("Set Scale...", "distance=2.87 known=1 unit=um");
run("Color Threshold...");
// Color Thresholder 1.49m
// Autogenerated macro, single images only!
min=newArray(3);
max=newArray(3);
filter=newArray(3);
a=getTitle();
run("HSB Stack");
run("Convert Stack to Images");
selectWindow("Hue");
rename("0");
selectWindow("Saturation");
rename("1");
selectWindow("Brightness");
rename("2");
min[0]=0;
max[0]=255;
filter[0]="pass";
min[1]=0;
max[1]=255;
filter[1]="pass";
min[2]=18;
max[2]=135;
filter[2]="pass";
for (i=0;i<3;i++){
 selectWindow(""+i);
 setThreshold(min[i], max[i]);
 run("Convert to Mask");
 if (filter[i]=="stop")  run("Invert");
}
imageCalculator("AND create", "0","1");
imageCalculator("AND create", "Result of 0","2");
for (i=0;i<3;i++){
 selectWindow(""+i);
 close();
}
selectWindow("Result of 0");
close();
selectWindow("Result of Result of 0");
rename(a);
// Colour Thresholding-------------
run("8-bit");
setAutoThreshold("Default");
//run("Threshold...");
setOption("BlackBackground", false);
run("Convert to Mask");
run("Analyze Particles...", "  show=Masks display summarize");
